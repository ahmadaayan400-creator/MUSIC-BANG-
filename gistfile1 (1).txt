local function stopAllSounds()
	pcall(function()
		sound1:Stop()
		sound2:Stop()
		sound3:Stop()
	end)
end

local function handleAudioForMode(currentMode)
	if soundMuted then
		stopAllSounds()
		return
	end
	if tick() - lastSoundTime < soundGap then return end
	if currentMode == "Normal" then
		if sound2.IsPlaying then sound2:Stop() end
		if sound3.IsPlaying then sound3:Stop() end
		if not sound1.IsPlaying then
			sound1:Play()
			lastSoundTime = tick()
		end
	elseif currentMode == "Face Bang" then
		if sound1.IsPlaying then sound1:Stop() end
		if sound3.IsPlaying then sound3:Stop() end
		if not sound2.IsPlaying then
			sound2:Play()
			lastSoundTime = tick()
		end
	elseif currentMode == "Auto Switch" then
		if sound1.IsPlaying then sound1:Stop() end
		if sound2.IsPlaying then sound2:Stop() end
		if not sound3.IsPlaying then
			sound3:Play()
			lastSoundTime = tick()
		end
	end
end

end

local function createSlider(minY, labelText, minVal, maxVal, defaultVal, callback)
	local label = Instance.new("TextLabel", mainFrame)
	label.Size = UDim2.new(1, -16, 0, 13)
	label.Position = UDim2.new(0, 8, 0, minY)
	label.Text = labelText .. ": " .. tostring(defaultVal)
	label.TextColor3 = ACCENT
	label.Font = Enum.Font.GothamBold
	label.TextSize = 10
	label.BackgroundTransparency = 1
	label.TextXAlignment = Enum.TextXAlignment.Left

	local track = Instance.new("TextButton", mainFrame)
	track.Size = UDim2.new(1, -16, 0, 4)
	track.Position = UDim2.new(0, 8, 0, minY + 14)
	track.BackgroundColor3 = Color3.fromRGB(40, 35, 10)
	track.Text = ""
	track.AutoButtonColor = false
	track.BorderSizePixel = 0
	Instance.new("UICorner", track).CornerRadius = UDim.new(0, 2)

	local fill = Instance.new("Frame", track)
	fill.Size = UDim2.new((defaultVal - minVal) / (maxVal - minVal), 0, 1, 0)
	fill.BackgroundColor3 = ACCENT
	fill.BorderSizePixel = 0
	Instance.new("UICorner", fill).CornerRadius = UDim.new(0, 2)

	local knob = Instance.new("Frame", track)
	knob.Size = UDim2.new(0, 8, 0, 10)
	knob.Position = UDim2.new((defaultVal - minVal) / (maxVal - minVal), -4, 0.5, -5)
	knob.BackgroundColor3 = Color3.fromRGB(255, 255, 200)
	knob.BorderSizePixel = 0
	Instance.new("UICorner", knob).CornerRadius = UDim.new(0, 3)

	local function updateSliderPosition(inputPositionX)
		local relativeX = math.clamp((inputPositionX - track.AbsolutePosition.X) / math.max(1, track.AbsoluteSize.X), 0, 1)
		knob.Position = UDim2.new(relativeX, -4, 0.5, -5)
		fill.Size = UDim2.new(relativeX, 0, 1, 0)
		local currentVal = math.floor((minVal + (relativeX * (maxVal - minVal))) * 10) / 10
		label.Text = labelText .. ": " .. tostring(currentVal)
		callback(currentVal)
	end

	local sliderDragging = false
	track.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			sliderDragging = true
			updateSliderPosition(input.Position.X)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if sliderDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			updateSliderPosition(input.Position.X)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			sliderDragging = false
		end
	end)
end

	if _G.EgoEmpireBangStorage.ActiveConnection then
		_G.EgoEmpireBangStorage.ActiveConnection:Disconnect()
		_G.EgoEmpireBangStorage.ActiveConnection = nil
	end
	if _G.EgoEmpireBangStorage.PlatformPart then
		_G.EgoEmpireBangStorage.PlatformPart:Destroy()
		_G.EgoEmpireBangStorage.PlatformPart = nil
	end
	if _G.EgoEmpireBangStorage.BV then
		_G.EgoEmpireBangStorage.BV:Destroy()
		_G.EgoEmpireBangStorage.BV = nil
	end
	if _G.EgoEmpireBangStorage.BG then
		_G.EgoEmpireBangStorage.BG:Destroy()
		_G.EgoEmpireBangStorage.BG = nil
	end
	if activeTrack then activeTrack:Stop() activeTrack = nil end
	stopAllSounds()
	toggleButton.Text = "> EXECUTE_BANG"
	toggleButton.BackgroundColor3 = Color3.fromRGB(0, 40, 10)
	toggleButton.TextColor3 = NAME_GREEN
	pcall(function()
		local char = player.Character
		if not char then return end
		local torso = char:FindFirstChild("Torso") or char:FindFirstChild("LowerTorso")
		local hrp = char:FindFirstChild("HumanoidRootPart")
		local rootJoint = (torso and torso:FindFirstChild("RootJoint")) or (hrp and hrp:FindFirstChild("RootJoint"))
		if rootJoint then rootJoint.C0 = CFrame.new() end
		local hum = char:FindFirstChildOfClass("Humanoid")
		if hum then hum.PlatformStand = false end
	end)
end

local function updateList()
	for _, child in pairs(scrollingFrame:GetChildren()) do
		if child:IsA("TextButton") then child:Destroy() end
	end
	local filter = textBox.Text:lower()
	if string.find(filter, "selected:") then filter = "" end
	local count = 0
	for _, p in pairs(Players:GetPlayers()) do
		if p ~= player then
			local match = filter == ""
				or string.find(p.Name:lower(), filter, 1, true)
				or string.find(p.DisplayName:lower(), filter, 1, true)
			if match then
				count = count + 1
				local selected = table.find(selectedPlayers, p) ~= nil
				local btn = Instance.new("TextButton")
				btn.Size = UDim2.new(1, -4, 0, 22)
				-- GREEN names (suggestion / list) — hacker style
				btn.Text = (selected and "[✓] " or "[ ] ") .. p.DisplayName .. " @" .. p.Name
				btn.BackgroundColor3 = selected and Color3.fromRGB(0, 40, 10) or Color3.fromRGB(0, 14, 4)
				btn.BackgroundTransparency = 0.2
				btn.TextColor3 = selected and NAME_SEL or NAME_GREEN -- selected = WHITE, else green
				btn.TextSize = 11
				btn.Font = Enum.Font.Code
				btn.TextXAlignment = Enum.TextXAlignment.Left
				btn.BorderSizePixel = 0
				btn.Parent = scrollingFrame
				Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 3)
				local rowStroke = Instance.new("UIStroke", btn)
				rowStroke.Color = selected and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(0, 60, 20)
				rowStroke.Thickness = 1
				rowStroke.Transparency = selected and 0.15 or 0.55

				btn.MouseButton1Click:Connect(function()
					local index = table.find(selectedPlayers, p)
					if index then
						table.remove(selectedPlayers, index)
					else
						table.insert(selectedPlayers, p)
					end
					refreshCounter()
					updateList()
				end)
			end
		end
	end
	scrollingFrame.CanvasSize = UDim2.new(0, 0, 0, math.max(count * 25, 72))
	refreshCounter()
end

				activeTrack = hum:LoadAnimation(anim)

			if bv and bv.Parent then bv.Velocity = Vector3.new(0, 0, 0) end
			return

		if tick() - lastTargetSwitchTime >= multiSwitchInterval or not selectedPlayer then
			lastTargetSwitchTime = tick()
			currentTargetIndex = currentTargetIndex + 1
			if currentTargetIndex > #selectedPlayers then currentTargetIndex = 1 end
			selectedPlayer = selectedPlayers[currentTargetIndex]
		end

		if not selectedPlayer or not selectedPlayer.Character then return end

		local char = player.Character
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if not hrp or not hum or not bv or not bg then return end

		local targetHRP = selectedPlayer.Character:FindFirstChild("HumanoidRootPart")
		local targetHead = selectedPlayer.Character:FindFirstChild("Head")
		if not targetHRP then return end

		local currentModeToRun = targetMode
		if targetMode == "Auto Switch" then
			if tick() - lastSwitchTime >= 1.0 then
				lastSwitchTime = tick()
				currentAutoSubMode = (currentAutoSubMode == "Normal") and "Face Bang" or "Normal"
			end
			currentModeToRun = currentAutoSubMode
		end

		handleAudioForMode(targetMode)

		if targetHRP.Position.Y < -60 or targetHRP.Position.Y < (workspace.FallenPartsDestroyHeight + 15) then
			bv.Velocity = Vector3.new(0, 0, 0)
			if platform and platform.Parent then
				platform.CFrame = hrp.CFrame * CFrame.new(0, -3.1, 0)
			end
			return
		end

		hum.PlatformStand = true

		local timeFactor = tick() * speedMultiplier
		local sineWave = math.sin(timeFactor)
		local absSine = math.abs(sineWave)
		local dynamicOffset = absSine * forwardBackRange
		local tiltAngle = -sineWave * 0.15

		local targetCFrame
		if currentModeToRun == "Face Bang" then
			local basePart = targetHead or targetHRP
			targetCFrame = basePart.CFrame
				* CFrame.new(0, fixedUpOffset, -attachDistance - dynamicOffset - 0.2)
				* CFrame.Angles(0, math.pi, 0)
				+ basePart.CFrame.LookVector * (sineWave * 0.7)
		else
			targetCFrame = targetHRP.CFrame
				* CFrame.new(0, 0, attachDistance + dynamicOffset + 0.8 + sineWave)
				* CFrame.Angles(tiltAngle + math.rad(sineWave * 15), 0, 0)
		end

		-- Sticky apply (+ hard sync)
		hrp.CFrame = targetCFrame
		bg.CFrame = targetCFrame
		bv.Velocity = (targetCFrame.Position - hrp.Position) * 300
		if hardSync then
			hrp.CFrame = targetCFrame
			hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
			hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
		end

		if platform and platform.Parent then
			platform.CFrame = targetCFrame * CFrame.new(0, -3.1, 0)
		end

		pcall(function()
			local torso = char:FindFirstChild("Torso") or char:FindFirstChild("LowerTorso")
			local rootJoint = (torso and torso:FindFirstChild("RootJoint")) or hrp:FindFirstChild("RootJoint")
			if rootJoint then
				local hipThrustAngle = math.rad(sineWave * (forwardBackRange * 15))
				local jointOffsetZ = -absSine * (forwardBackRange * 0.2)
				rootJoint.C0 = CFrame.new(0, 0, jointOffsetZ)
					* CFrame.Angles((currentModeToRun == "Face Bang" and hipThrustAngle or -hipThrustAngle), 0, 0)
			end
		end)

		pcall(function()
			for _, v in pairs(char:GetDescendants()) do
				if v:IsA("BasePart") then
					v.CanCollide = false
				end
			end
		end)
	end)

	notify("EGO EMPIRE", "Bang started · " .. tostring(#selectedPlayers) .. " target(s)", 3)
end)
